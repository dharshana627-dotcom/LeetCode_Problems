class Solution {
    public int lengthOfLIS(int[] nums) {
        if (nums == null || nums.length == 0) 
            return 0;
        ArrayList<Integer> sub = new ArrayList<>();
        for (int num:nums) {
            int i = Collections.binarySearch(sub, num);
            if (i<0) {
                i = -(i+1);
            }
            if (i==sub.size()) {
                sub.add(num);
            } else {
                sub.set(i, num);
            }
        }
        return sub.size();
    }
}
