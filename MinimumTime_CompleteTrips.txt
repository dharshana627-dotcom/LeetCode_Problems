class Solution {
    public long minimumTime(int[] time, int totalTrips) {
        long start = 1, end = Long.MAX_VALUE, ans = 0, mid = 0;
        long minTime = time[0];
        for (int t : time) {
            minTime = Math.min(minTime, t);
        }
        end = minTime * totalTrips;
        while (start <= end) {
            mid = start + (end - start) / 2;
            if (canComplete(time, totalTrips, mid)) {
                end = mid - 1;
                ans = mid;
            } else {
                start = mid + 1;
            }
        } 
        return ans; 
    } 
    public boolean canComplete(int[] arr, int trip, long mid) {
        long sum = 0; 
        for (int each : arr) {
            sum += (mid / each);
            if (sum >= trip) return true;
        } 
        return sum >= trip; 
    } 
}
