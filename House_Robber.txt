class Solution {
    public int rob(int[] nums) {
       if(nums.length==0){
        return 0;
       }
       if(nums.length==1){
        return nums[0];
       }
       int p2=nums[0],p1=Math.max(nums[0],nums[1]),c=0;
       for(int i=2;i<nums.length;i++){
        c=Math.max(p1,(nums[i]+p2));
        p2=p1;p1=c;
       }
       return p1;
    }
}
