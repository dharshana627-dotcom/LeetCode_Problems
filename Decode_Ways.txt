class Solution {
    public int numDecodings(String s) {
        if(s==null||s.length()==0)
            return 0;
        int n=s.length(),ind=0,pair=0;
        int[]dp=new int[n+1];
        dp[0]=1;dp[1]=(s.charAt(0)!='0')?1:0;
        for(int i=2;i<=n;i++){
            ind=s.charAt(i-1)-'0';
            if(ind>=1 && ind<=9){
                dp[i]+=dp[i-1];
            }
            String sub=s.substring(i-2,i);
            pair=Integer.parseInt(sub);
            if(pair>=10 && pair<=26){
                dp[i]+=dp[i-2];
            }
        }
        return dp[n];
    }
}
