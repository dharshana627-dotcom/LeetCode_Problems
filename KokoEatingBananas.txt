class Solution {
    public int minEatingSpeed(int[] piles, int h) {
        int low = 1,high = 0;
        for (int p : piles) {
            high=Math.max(high, p);
        }
        int res=high;
        while (low<=high) {
            int mid = low + (high-low)/2;
            if (canEatAll(piles, h, mid)) {
                res=mid;
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }   
        return res;
    }
    public static boolean canEatAll(int[] piles, int h, int k) {
        long total=0;
        for (int p : piles) {
            total+=(p+k-1) / k;
        }
        return total<=h;
    }
}
